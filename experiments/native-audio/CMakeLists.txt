cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(LaokuaiNativeAudioLab LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
  message(FATAL_ERROR "This experimental host currently supports Windows only")
endif()
set(AOO_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.local/native-audio-deps/aoo" CACHE PATH "Pinned AoO checkout")
foreach(flag AOO_BUILD_EXAMPLES AOO_BUILD_DOCUMENTATION AOO_BUILD_PD_EXTERNAL AOO_BUILD_SC_EXTENSION AOO_BUILD_SERVER AOO_BUILD_SHARED_LIBRARY AOO_BUILD_TESTS AOO_INSTALL_LIBRARY AOO_INSTALL_PKG_CONFIG_MODULE AOO_INSTALL_CMAKE_CONFIG_MODULE AOO_NET AOO_USE_IPV6)
  set(${flag} OFF CACHE BOOL "Lab configuration" FORCE)
endforeach()
set(AOO_USE_OPUS ON CACHE BOOL "" FORCE)
set(AOO_LOCAL_OPUS ON CACHE BOOL "" FORCE)
set(PA_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(PA_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(PA_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
foreach(flag PA_USE_ASIO PA_USE_DS PA_USE_WDMKS PA_BUILD_TESTS PA_BUILD_EXAMPLES)
  set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(PA_USE_WASAPI ON CACHE BOOL "" FORCE)
set(PA_USE_WMME ON CACHE BOOL "" FORCE)
# Pinned PortAudio predates CMake 4's removal of pre-3.5 policy compatibility.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
add_subdirectory("${AOO_SOURCE_DIR}" aoo)
add_subdirectory("${AOO_SOURCE_DIR}/deps/portaudio" portaudio)
add_library(native_audio_engine STATIC src/audio_engine.cpp)
target_include_directories(native_audio_engine PUBLIC src)
target_link_libraries(native_audio_engine PUBLIC Aoo::aoo)
target_compile_definitions(native_audio_engine PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(laokuai-native-audio src/main.cpp)
target_link_libraries(laokuai-native-audio PRIVATE native_audio_engine portaudio_static ws2_32 bcrypt)
target_include_directories(laokuai-native-audio PRIVATE "${AOO_SOURCE_DIR}/deps/portaudio/include")
target_compile_definitions(laokuai-native-audio PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(native-audio-offline tests/offline.cpp)
target_link_libraries(native-audio-offline PRIVATE native_audio_engine)
foreach(target native_audio_engine laokuai-native-audio native-audio-offline)
  target_compile_options(${target} PRIVATE /W4 /utf-8)
endforeach()
enable_testing()
add_test(NAME native-audio-offline COMMAND native-audio-offline)
add_test(NAME native-audio-host-self-test COMMAND laokuai-native-audio --self-test)
